Alpine.js PoC — Live filter project list

เทียบ 2 วิธีใช้งาน feature เดียวกัน (ค้นหาโปรเจคจากชื่อ) — ข้อมูลจริงจาก /api/projects

Vanilla JS current pattern

กำลังโหลด...
// --- code ~18 lines ---
const vInput = document.getElementById('v-search');
const vList  = document.getElementById('v-list');
const vCount = document.getElementById('v-count');
let vAll = [];

function vRender() {
  const q = vInput.value.toLowerCase();
  const filtered = vAll.filter(p => p.name.toLowerCase().includes(q));
  vCount.textContent = `${filtered.length} / ${vAll.length} projects`;
  vList.innerHTML = filtered.length
    ? filtered.map(p => `<div class="item"><span class="name">${p.name}</span><span class="type">${p.type}</span></div>`).join('')
    : '<div class="empty">ไม่พบโปรเจค</div>';
}

fetch('/api/projects').then(r => r.json()).then(list => { vAll = list; vRender(); });
vInput.addEventListener('input', vRender);

Alpine.js reactive pattern

ไม่พบโปรเจค
<!-- code ~9 lines (in HTML, no separate script) -->
<div x-data="{
    all: [],
    q: '',
    get filtered() { return this.all.filter(p => p.name.toLowerCase().includes(this.q.toLowerCase())); }
  }"
  x-init="all = await (await fetch('/api/projects')).json()">

  <input x-model="q" placeholder="พิมพ์ค้นหา...">
  <div x-text="`${filtered.length} / ${all.length} projects`"></div>
  <template x-for="p in filtered" :key="p.id">
    <div><span x-text="p.name"></span></div>
  </template>
</div>
